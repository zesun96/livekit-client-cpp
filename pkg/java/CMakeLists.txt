cmake_minimum_required(VERSION 3.20)
project(livekit_client_java LANGUAGES CXX)

set(JAVA_JNI_INCLUDE_DIR "" CACHE PATH "Directory containing jni.h and jni_md.h (optional when a JDK is installed)")
if(JAVA_JNI_INCLUDE_DIR)
  set(_jni_includes "${JAVA_JNI_INCLUDE_DIR}")
else()
  find_package(JNI REQUIRED COMPONENTS JVM)
  set(_jni_includes ${JNI_INCLUDE_DIRS})
endif()
set(LIVEKIT_CLIENT_CPP_LIBRARY "" CACHE FILEPATH "Path to livekitclient shared library")
if(NOT EXISTS "${LIVEKIT_CLIENT_CPP_LIBRARY}")
  message(FATAL_ERROR "Set LIVEKIT_CLIENT_CPP_LIBRARY to a built livekitclient shared library")
endif()

add_library(livekitclient_java SHARED src/main/cpp/livekit_jni.cpp)
target_compile_features(livekitclient_java PRIVATE cxx_std_20)
target_include_directories(livekitclient_java PRIVATE ${_jni_includes} ../../include)
target_compile_definitions(livekitclient_java PRIVATE LKC_SHARED)
target_link_libraries(livekitclient_java PRIVATE ${LIVEKIT_CLIENT_CPP_LIBRARY})
