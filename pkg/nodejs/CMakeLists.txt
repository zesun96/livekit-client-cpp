cmake_minimum_required(VERSION 3.20)
project(livekit_client_node LANGUAGES CXX)
option(LIVEKIT_NODE_TESTING "Compile private callback injection for offline tests only" OFF)

set(NODE_INCLUDE_DIR "" CACHE PATH "Directory containing node_api.h")
set(NODE_LIBRARY "" CACHE FILEPATH "Node import library (Windows)")
set(LIVEKIT_CLIENT_CPP_LIBRARY "" CACHE FILEPATH "Built livekitclient shared import library")
if(NOT EXISTS "${NODE_INCLUDE_DIR}/node_api.h" OR NOT EXISTS "${LIVEKIT_CLIENT_CPP_LIBRARY}")
  message(FATAL_ERROR "Set NODE_INCLUDE_DIR and LIVEKIT_CLIENT_CPP_LIBRARY to existing development files")
endif()
add_library(livekitclient_node MODULE src/addon.cpp)
target_compile_features(livekitclient_node PRIVATE cxx_std_20)
target_compile_definitions(livekitclient_node PRIVATE NAPI_VERSION=8 LKC_SHARED)
if(LIVEKIT_NODE_TESTING)
  target_compile_definitions(livekitclient_node PRIVATE LIVEKIT_NODE_TESTING)
endif()
target_include_directories(livekitclient_node PRIVATE "${NODE_INCLUDE_DIR}" ../../include)
target_link_libraries(livekitclient_node PRIVATE "${LIVEKIT_CLIENT_CPP_LIBRARY}")
set_target_properties(livekitclient_node PROPERTIES PREFIX "" SUFFIX ".node")
if(WIN32)
  if(NOT EXISTS "${NODE_LIBRARY}")
    message(FATAL_ERROR "Set NODE_LIBRARY to the matching Windows node.lib")
  endif()
  target_link_libraries(livekitclient_node PRIVATE "${NODE_LIBRARY}")
endif()
if(MSVC)
  target_compile_options(livekitclient_node PRIVATE /W4 /permissive- /utf-8)
  set_property(TARGET livekitclient_node PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(APPLE)
  target_link_options(livekitclient_node PRIVATE "-undefined" "dynamic_lookup")
endif()
